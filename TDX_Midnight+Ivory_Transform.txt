-- TDX / MIDNIGHT + IVORY
-- One application, one gameplay worker, two independently constructed views.
-- Uses the exact TDX client contracts supplied in the original scripts.
-- Reviewed statically; not runtime-tested inside Roblox/TDX.
--
-- The reveal uses temporary, pruned horizontal GUI slices. Each slice clips
-- an incoming-view copy at a calculated diagonal boundary. These copies have
-- no callbacks, workers, or event connections. Shared display updates are
-- mirrored into them while the transition runs.
--
-- Manual theme requests can reverse an in-progress reveal without resetting
-- gameplay, rebuilding the views, or starting another worker.

local function existingProblem()
    for _, name in ipairs(OLD_NAMES) do
        for _, old in ipairs(playerGui:GetChildren()) do
            if old.Name == name then
                local pending = old:GetAttribute("OutstandingUpgrade")
                if pending == true then
                    return "Existing panel has an unresolved upgrade."
                elseif pending == nil then
                    return "Close the older panel after its upgrades settle."
                end
            end
        end
    end
end

local function positive(value, fallback, minimum)
    if type(value) ~= "number" or value ~= value or math.abs(value) == math.huge then
        return fallback
    end
    return math.max(minimum, value)
end

local function connect(signal, callback)
    local connection = signal:Connect(callback)
    S.connections[#S.connections + 1] = connection
    return connection

local function make(className, parent, properties)
    local object = Instance.new(className)
    for key, value in pairs(properties or {}) do
        object[key] = value
    end
    object.Parent = parent
    return object
end

local function corner(object, radius)
    return make("UICorner", object, {
        CornerRadius = UDim.new(0, radius or 12),
    })
end

local function stroke(object, color, transparency, thickness)
    return make("UIStroke", object, {
        Color = color,
        Transparency = transparency or 0,
        Thickness = thickness or 1,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    })
end

local function gradient(object, first, last, rotation)
    return make("UIGradient", object, {
        Color = ColorSequence.new(first, last),
        Rotation = rotation or 0,
    })
end

local function set(object, property, value)
    if object[property] ~= value then
        object[property] = value
    end
    local copies = T.mirrors[object]
    if copies then
        for _, copy in ipairs(copies) do
            if copy[property] ~= value then
                copy[property] = value
            end
        end
    end
end

local function replaceTween(key, object, info, goals)
    local old = visualTweens[key]
    if old then
        visualTweens[key] = nil
        old:Cancel()
        old:Destroy()
    end
    local tween = TweenService:Create(object, info, goals)
    visualTweens[key] = tween
    tween:Play()
    return tween
end

local function button(v, parent, value, position, size, command, argument)
    local object = make("TextButton", parent, {
        Position = position,
        Size = size,
        BackgroundColor3 = v.C.surface,
        BorderSizePixel = 0,
        Text = value,
        TextColor3 = v.C.text,
        TextSize = 12,
        Font = v.name == "MIDNIGHT" and Enum.Font.GothamBold or Enum.Font.GothamMedium,
        TextWrapped = true,
        AutoButtonColor = false,
        ZIndex = 4,
    })
    corner(object, v.name == "MIDNIGHT" and 9 or 12)
    local outline = stroke(object, v.C.border, 0.25)
    local feedback = stroke(object, v.C.secondary, 1, 2)
    local scale = make("UIScale", object, {Scale = 1})
    local record = {
        object = object,
        scale = scale,
        outline = feedback,
        hovered = false,
    }
    feedbackRecords[#feedbackRecords + 1] = record

    local function animate(pressed)
        if not S.alive then return end
        replaceTween(scale, scale, TweenInfo.new(0.14), {
            Scale = pressed and 0.982 or 1,
        })
        replaceTween(feedback, feedback, TweenInfo.new(0.16), {
            Transparency = pressed and 0.1 or (record.hovered and 0.35 or 1),
        })
    end

    connect(object.MouseEnter, function()
        record.hovered = true
        animate(false)
    end)
    connect(object.MouseLeave, function()
        record.hovered = false
        animate(false)
    end)
    connect(object.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            animate(true)
        end
    end)
    connect(object.InputEnded, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            animate(false)
        end
    end)
    connect(object.Activated, function()
        if dispatch then dispatch(command, argument) end
    end)
    return object, outline
end

local function resetFeedback()
    for _, record in ipairs(feedbackRecords) do
        for _, key in ipairs({record.scale, record.outline}) do
            local tween = visualTweens[key]
            if tween then
                visualTweens[key] = nil
                tween:Cancel()
                tween:Destroy()
            end
        end
        record.hovered = false
        record.scale.Scale = 1
        record.outline.Transparency = 1
    end
end

local function addLighting(v)
    local dark = v.name == "MIDNIGHT"
    local ambient = make("Frame", v.root, {
        Name = "Atmosphere",
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BackgroundTransparency = dark and 0.70 or 0.12,
        BorderSizePixel = 0,
        Active = false,
        Interactable = false,
        ZIndex = 1,
    })
    corner(ambient, dark and 16 or 24)

    local ambientGradient = make("UIGradient", ambient, {
        Rotation = dark and 32 or 24,
        Offset = Vector2.new(-1.2, -0.12),
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, v.C.secondary),
            ColorSequenceKeypoint.new(0.5,
                dark and Color3.fromRGB(201, 226, 255)
                    or Color3.fromRGB(255, 253, 225)),
            ColorSequenceKeypoint.new(1, v.C.accent),
        }),
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 1),
            NumberSequenceKeypoint.new(0.28, 1),
            NumberSequenceKeypoint.new(0.42, 0.90),
            NumberSequenceKeypoint.new(0.5, dark and 0.20 or 0.70),
            NumberSequenceKeypoint.new(0.58, 0.90),
            NumberSequenceKeypoint.new(0.72, 1),
            NumberSequenceKeypoint.new(1, 1),
        }),
    })

    local tween = TweenService:Create(
        ambientGradient,
        TweenInfo.new(dark and 14 or 13, Enum.EasingStyle.Sine,
            Enum.EasingDirection.InOut, -1, true, 0.5),
        {Offset = Vector2.new(1.2, 0.16)}
    )
    visualTweens[ambientGradient] = tween
    v.lights[#v.lights + 1] = tween

    if dark then
        -- These HUD rails are decorative background elements. Mark them so
        -- the cinematic slice renderer does not clone them once per slice;
        -- doing so would multiply the rails into the visible line pattern
        -- during the reveal. They remain part of the normal Midnight view.
        for i = 1, 4 do
            local rail = rule(v.root, UDim2.new(i / 5, 0, 0, 0),
                UDim2.new(0, 1, 1, 0), v.C.accent, 0.96)
            rail.ZIndex = 1
            rail.Name = "HUDRailV" .. i
            rail:SetAttribute("TransitionSkip", true)
        end
        for i = 1, 5 do
            local rail = rule(v.root, UDim2.new(0, 0, i / 6, 0),
                UDim2.new(1, 0, 0, 1), v.C.accent, 0.975)
            rail.ZIndex = 1
            rail.Name = "HUDRailH" .. i
            rail:SetAttribute("TransitionSkip", true)
        end
    else
        rule(v.root, UDim2.new(1, -30, 0, 103),
            UDim2.new(0, 1, 1, -120), v.C.secondary, 0.86)
    end
end

local function message(value, bad)
    if not S.alive then return end
    S.status, S.bad = value, bad == true
end

local function outstanding()
    S.outstanding = S.uncertain ~= nil or (S.job ~= nil and S.job.pending ~= nil)
    gui:SetAttribute("OutstandingUpgrade", S.outstanding)
end

local function current()
    local tower = API.UI.GetCurrentTower()
    if tower and API.Towers.GetTower(tower.Hash) == tower then
        return tower
    end
end

local function cash()
    local wrapper = API.Players.GetLocalPlayerWrapper()
    local value = wrapper and wrapper.Stats and wrapper.Stats:GetCash()
    if value == nil then return 0 end
    assert(type(value) == "number" and value == value
        and math.abs(value) < math.huge, "Invalid client cash.")
    return value
end

local function selectionProblem(tower)
    if not tower then return "Select a tower first." end
    if API.Towers.GetTower(tower.Hash) ~= tower then
        return "Tower no longer exists."
    end
    if not tower.OwnedByLocalPlayer then return "This tower is not yours." end
    if workspace:GetAttribute("FPSMode") or API.Input.GetState() ~= "Normal" then
        return "Finish placement, aiming, or direct control."
    end
    if not tower:Alive() then return "Tower is not alive." end
    if tower:IsRebuilding() then return "Tower is rebuilding." end
    if tower:IsReviving() then return "Tower is reviving." end
end

local function upgradeProblem(tower)
    local why = selectionProblem(tower)
    if why then return why end
    if S.uncertain then
        return "Previous outcome unknown. Waiting for a level update."
    end
    if tower.AbilityHandler and not tower.AbilityHandler:CanUpgrade() then
        return "Cannot upgrade during this ability."
    end
end

local function nextStep(tower, path)
    local lh = tower.LevelHandler
    local data = lh:GetUpgradePathData()
    local a, b = lh:GetLevelOnPath(1), lh:GetLevelOnPath(2)

    if data.OnePathOnly then
        return nil, "This preset needs a two-path tower."
    end
    if a > path.a or b > path.b then
        return nil, "Cannot downgrade to this preset."
    end
    if path.a > #data.Path1Data or path.b > #data.Path2Data then
        return nil, "Preset unavailable."
    end

    local lock = lh:GetPathLockLevel()
    if path.a >= lock and path.b >= lock then
        return nil, "Paths conflict."
    end
    local major = path.a > path.b and 1 or 2
    local lockedPath = lh:GetPath()
    if lockedPath ~= 0 and lockedPath ~= major then
        return nil, "Opposite path locked."
    end
    if a == path.a and b == path.b then
        return nil, nil, true
    end

    local side, before
    if major == 1 then
        if b < path.b then
            side, before = 2, b
        else
            side, before = 1, a
        end
    else
        if a < path.a then
            side, before = 1, a
        else
            side, before = 2, b
        end
    end

    if before + 1 > lh:GetMaxPossibleLevel(side) then
        return nil, "Path locked."
    end

    local cost = lh:GetLevelUpgradeCost(side, 1,
        tower.BuffHandler:GetDiscount(),
        API.Game.GetTowerCostMultiplier(tower.Type),
        tower:GetDynamicPriceScalingData())

    if type(cost) ~= "number" or cost ~= cost or cost < 0 or cost == math.huge then
        return nil, "Price unavailable."
    end
    return {side = side, level = before + 1, cost = cost}
end

local function finish(value, bad)
    S.job = nil
    outstanding()
    message(value, bad)
end

local function tickUncertain()
    local record = S.uncertain
    if not record then return end
    local tower, pending = record.tower, record.pending

    if API.Towers.GetTower(tower.Hash) ~= tower then
        S.uncertain = nil
        outstanding()
        message("Tower removed. Unresolved tower request cleared.")
    elseif tower.LevelHandler:GetLevelOnPath(pending.side) >= pending.level then
        S.uncertain = nil
        outstanding()
        message("Late level update received. Review before restarting.")
    end
end

local function tickJob()
    local job = S.job
    if not job then return end
    local tower = job.tower

    if API.Towers.GetTower(tower.Hash) ~= tower then
        finish("Tower removed; stopped.", true)
        return
    end
    if API.UI.GetCurrentTower() ~= tower then
        job.cancelReason = job.cancelReason or "Selection changed."
    end

    local pending = job.pending
    if pending then
        if tower.LevelHandler:GetLevelOnPath(pending.side) >= pending.level then
            job.pending = nil
            outstanding()
        elseif os.clock() >= pending.deadline then
            S.uncertain = {tower = tower, pending = pending}
            S.auto = false
            table.clear(S.fresh)
            finish("Outcome UNKNOWN. Auto stopped; no retry issued.", true)
            return
        else
            return
        end
    end

    if job.cancelReason then
        finish(job.cancelReason .. " Completed purchases remain.")
        return
    end

    local why = upgradeProblem(tower)
    if why then
        finish(why, true)
        return
    end

    local step, stepError, done = nextStep(tower, job.path)
    if done then
        finish(string.format("Build complete. Reached %d-%d.", job.path.a, job.path.b))
        return
    end
    if not step then
        finish(stepError, true)
        return
    end

    local balance = cash()
    if balance < step.cost then
        message(string.format("Waiting for %d more cash.", math.ceil(step.cost - balance)))
        return
    end
    if API.UI.GetCurrentTower() ~= tower then
        finish("Selection changed.")
        return
    end

    job.pending = {
        side = step.side,
        level = step.level,
        deadline = os.clock() + ACK_TIMEOUT,
    }
    outstanding()
    API.Upgrade:Fire(step.side, false)
    message(string.format("Path %d -> %d. Waiting for level update.", step.side, step.level))
end

local function guarded(callback)
    if not S.alive or S.busy or S.faulted then return end
    S.busy = true
    local ok, failure = xpcall(callback, debug.traceback)
    S.busy = false
    if ok then return end

    if S.job and S.job.pending then
        S.uncertain = {tower = S.job.tower, pending = S.job.pending}
    end
    S.job, S.auto, S.targetAuto, S.faulted = nil, false, false, true
    table.clear(S.fresh)
    outstanding()
    message("Client contract error. Worker stopped; see console.", true)
    warn("[TDX] " .. tostring(failure))
end

local function start(fromAuto)
    if S.job or S.uncertain or S.closing then return end
    local tower = current()
    local why = upgradeProblem(tower)
    if why then
        message(why, true)
        return
    end
    S.fresh[tower] = nil
    S.job = {tower = tower, path = PATHS[S.path]}
    message(fromAuto and "Auto started." or "Started.")
    tickJob()
end

local scanElapsed = 0

local function observeTower(tower, now)
    if not tower or S.seen[tower] then return end
    S.seen[tower] = true
    if S.auto and tower.OwnedByLocalPlayer then
        S.fresh[tower] = now + AUTO_WINDOW
    end
end

local function tickAuto(dt)
    if not S.auto or S.closing or S.uncertain then return end
    local now, tower = os.clock(), current()
    observeTower(tower, now)
    scanElapsed = scanElapsed + dt

    if scanElapsed >= 0.15 then
        scanElapsed = 0
        for _, item in pairs(API.Towers.GetTowers()) do
            observeTower(item, now)
        end
        for item in pairs(S.seen) do
            if API.Towers.GetTower(item.Hash) ~= item then
                S.seen[item], S.fresh[item] = nil, nil
            end
        end
        for item, deadline in pairs(S.fresh) do
            if now > deadline then S.fresh[item] = nil end
        end
    end

    if tower and not S.job then
        local deadline = S.fresh[tower]
        if deadline and now <= deadline and not upgradeProblem(tower) then
            S.fresh[tower] = nil
            start(true)
        end
    end
end

local function tickTargeting()
    if not S.targetAuto or S.closing or S.uncertain or S.outstanding then return end
    local tower = current()
    if not tower or S.targeted[tower] or selectionProblem(tower) then return end
    S.targeted[tower] = true
    API.Target:Fire()
end

local function money(value)
    local result = tostring(math.floor(value))
    local count
    repeat
        result, count = result:gsub("^(-?%d+)(%d%d%d)", "%1,%2")
    until count == 0
    return "$" .. result
end

        owned = tower ~= nil and tower.OwnedByLocalPlayer == true,

        cash = balance,
    }
end

local function updateThemeButtons()
    local destination = T.current
    if T.transition then
        destination = T.transition.to == 1
            and T.transition.incoming or T.transition.outgoing
    end
    for name, object in pairs(themeButtons) do
        local selected = name == "AUTO" and T.automatic
            or name ~= "AUTO" and not T.automatic and name == destination
        object.BackgroundColor3 = selected
            and Color3.fromRGB(67, 88, 111) or Color3.fromRGB(31, 42, 56)
        object.TextTransparency = THEME_SETTINGS.Enabled and 0 or 0.45
    end
end

local function resetThemeDeadline()
    T.deadline = os.clock() + (T.current == "MIDNIGHT"
        and THEME_SETTINGS.MidnightDuration or THEME_SETTINGS.IvoryDuration)
end

local function syncLighting()
    for name, v in pairs(views) do
        local active = not S.collapsed and not T.transition and name == T.current
        for _, tween in ipairs(v.lights) do
            if active then tween:Play() else tween:Pause() end
        end
    end
end

local function destroyTransition()
    local tr = T.transition
    T.transition = nil
    table.clear(T.mirrors)
    if tr and tr.layer then tr.layer:Destroy() end
    for name, v in pairs(views) do
        v.root.Visible = name == T.current
        v.body.ScrollingEnabled = true
    end
end

local function settleTransition()
    local tr = T.transition
    if not tr then return end
    T.current = tr.to == 1 and tr.incoming or tr.outgoing
    destroyTransition()
    resetThemeDeadline()
    updateThemeButtons()
    syncLighting()
end

-- Clone only an object's own properties. Children are visited selectively
-- afterward, so every slice excludes unrelated off-slice UI branches.
local function shallowClone(source)
    local children = source:GetChildren()
    local saved = {}
    for i, item in ipairs(children) do
        saved[i] = item.Archivable
        item.Archivable = false
    end
    local originalArchivable = source.Archivable
    source.Archivable = true
    local ok, copy = pcall(function() return source:Clone() end)
    source.Archivable = originalArchivable
    for i, item in ipairs(children) do
        item.Archivable = saved[i]
    end
    if not ok then error(copy) end
    assert(copy, "Could not create transition slice.")
    return copy, children
end

local function cloneSliceBranch(source, parent, top, bottom, clipTop, clipBottom)
    -- Decorative elements explicitly marked as transition-excluded should
    -- stay out of per-slice clones. This prevents full-panel rails/grids from
    -- being duplicated once for every reveal slice.
    if source:GetAttribute("TransitionSkip") then return nil end

    if source:IsA("GuiObject") then
        if not source.Visible then return nil end
        local y = source.AbsolutePosition.Y
        local h = source.AbsoluteSize.Y
        if y + h < top - 3 or y > bottom + 3
            or y + h < clipTop or y > clipBottom then
            return nil
        end
    end

    local copy, children = shallowClone(source)
    copy.Parent = parent

    local copies = T.mirrors[source]
    if not copies then
        copies = {}
        T.mirrors[source] = copies
    end
    copies[#copies + 1] = copy

    local nextTop, nextBottom = clipTop, clipBottom
    if source:IsA("GuiObject") then
        copy.Active = false
        copy.Interactable = false
        copy.Selectable = false
        if source.ClipsDescendants or source:IsA("ScrollingFrame") then
            nextTop = math.max(nextTop, source.AbsolutePosition.Y)
            nextBottom = math.min(nextBottom,
                source.AbsolutePosition.Y + source.AbsoluteSize.Y)
        end
    end

    for _, item in ipairs(children) do
        if item:IsA("GuiObject") then
            if nextBottom >= top and nextTop <= bottom then
                cloneSliceBranch(item, copy, top, bottom, nextTop, nextBottom)
            end
        elseif not item:IsA("LuaSourceContainer") then
            -- UIStroke, UICorner, UIGradient, UIScale and other UI components
            -- have no event handlers in their clones.
            local component = item:Clone()
            if component then
                component.Parent = copy
                local componentCopies = T.mirrors[item]
                if not componentCopies then
                    componentCopies = {}
                    T.mirrors[item] = componentCopies
                end
                componentCopies[#componentCopies + 1] = component
            end
        end
    end
    return copy
end

local function drawTransition(tr)
    local width, height = T.width, T.height
    local skew = math.min(110, width * 0.24)
    local margin = 52
    local center = -skew / 2 - margin
        + tr.p * (width + skew + margin * 2)

    for _, slice in ipairs(tr.slices) do
        local boundary = center + skew * (slice.middle / height - 0.5)
        slice.mask.Size = UDim2.fromOffset(
            math.clamp(boundary, 0, width), slice.height)
    end

    -- This diagonal beam covers only the small stair-step approximation at
    -- the mask edge. The mask, not beam transparency, reveals the new view.
    tr.beam.Position = UDim2.fromOffset(center, height / 2)
    tr.beam.Rotation = -math.deg(math.atan(skew / height))
end

local function beginTransition(destination)
    if destination == T.current or T.transition or not S.alive then return end
    resetFeedback()
    dragInput = nil

    local tr = {
        outgoing = T.current,
        incoming = destination,
        p = 0,
        from = 0,
        to = 1,
        started = os.clock(),
        duration = THEME_SETTINGS.TransitionDuration,
        slices = {},
    }
    T.transition = tr
    table.clear(T.mirrors)
    syncLighting()

    tr.layer = make("CanvasGroup", viewHost, {
        Name = "CinematicReveal",
        Size = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1,
        ClipsDescendants = true,
        GroupTransparency = 0,
        Active = true,
        ZIndex = 20,
    })

    local incoming = views[destination]
    local outgoing = views[T.current]
    incoming.root.Visible = true
    incoming.root.ZIndex = 1
    outgoing.root.Visible = true
    outgoing.root.ZIndex = 2

    for _, v in pairs(views) do v.body.ScrollingEnabled = false end

    local count = math.clamp(math.ceil(T.height / 12), 8, 72)
    local originY = incoming.root.AbsolutePosition.Y

    for index = 1, count do
        local y0 = math.floor((index - 1) * T.height / count)
        local y1 = math.floor(index * T.height / count)
        local h = y1 - y0

        local mask = make("Frame", tr.layer, {
            Name = "RevealSlice",
            Position = UDim2.fromOffset(0, y0),
            Size = UDim2.fromOffset(0, h),
            BackgroundTransparency = 1,
            BorderSizePixel = 0,
            ClipsDescendants = true,
            Active = false,
            Interactable = false,
            ZIndex = 1,
        })

        local physicalTop = originY + y0 * T.scale
        local physicalBottom = originY + y1 * T.scale
        local copy = cloneSliceBranch(incoming.root, mask,
            physicalTop, physicalBottom,
            originY, originY + T.height * T.scale)

        assert(copy, "Incoming view has no measurable transition bounds.")
        copy.AnchorPoint = Vector2.new(0, 0)
        copy.Position = UDim2.fromOffset(0, -y0)
        copy.Size = UDim2.fromOffset(T.width, T.height)
        copy.Visible = true
        copy.ZIndex = 1

        tr.slices[#tr.slices + 1] = {
            mask = mask,
            height = h,
            middle = (y0 + y1) / 2,
        }
    end

    incoming.root.Visible = false

    tr.beam = make("Frame", tr.layer, {
        Name = "DiagonalBoundaryLight",
        AnchorPoint = Vector2.new(0.5, 0.5),
        Size = UDim2.fromOffset(76, T.height * 1.8 + 240),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BorderSizePixel = 0,
        Active = false,
        Interactable = false,
        ZIndex = 10,
    })

    make("UIGradient", tr.beam, {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Color3.fromRGB(130, 168, 255)),
            ColorSequenceKeypoint.new(0.40, Color3.fromRGB(216, 232, 255)),
            ColorSequenceKeypoint.new(0.50, Color3.fromRGB(255, 255, 245)),
            ColorSequenceKeypoint.new(0.60, Color3.fromRGB(255, 241, 202)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(191, 208, 246)),
        }),
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 1),
            NumberSequenceKeypoint.new(0.20, 0.97),
            NumberSequenceKeypoint.new(0.36, 0.80),
            NumberSequenceKeypoint.new(0.46, 0.32),
            NumberSequenceKeypoint.new(0.49, 0.02),
            NumberSequenceKeypoint.new(0.51, 0.02),
            NumberSequenceKeypoint.new(0.54, 0.32),
            NumberSequenceKeypoint.new(0.64, 0.80),
            NumberSequenceKeypoint.new(0.80, 0.97),
            NumberSequenceKeypoint.new(1, 1),
        }),
    })

    -- Input shield prevents interaction with visible-but-partially-clipped
    -- controls. The separate theme bar remains interactive for reversal.
    make("TextButton", tr.layer, {
        Name = "TransitionInputShield",
        Size = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1,
        Text = "",
        AutoButtonColor = false,
        Active = true,
        Selectable = false,
        ZIndex = 30,
    })

    tr.started = os.clock()
    drawTransition(tr)
    updateThemeButtons()
end

local function visualFailure(failure)
    local tr = T.transition
    if tr then T.current = tr.outgoing end
    destroyTransition()
    T.automatic = false
    updateThemeButtons()
    syncLighting()
    warn("[TDX] Theme transition stopped: " .. tostring(failure))
end

requestTheme = function(mode)
    if not S.alive or not THEME_SETTINGS.Enabled then return end
    if mode ~= "MIDNIGHT" and mode ~= "IVORY" and mode ~= "AUTO" then return end

    if mode == "AUTO" then
        T.automatic = true
        resetThemeDeadline()
        updateThemeButtons()
        return
    end

    T.automatic = false
    local tr = T.transition
    if tr then
        local goal = mode == tr.incoming and 1 or 0
        if goal ~= tr.to then
            tr.from = tr.p
            tr.to = goal
            tr.started = os.clock()
            tr.duration = math.max(0.12,
                THEME_SETTINGS.TransitionDuration * math.abs(goal - tr.p))
        end
        updateThemeButtons()
        return
    end

    resetThemeDeadline()
    local ok, failure = xpcall(function()
        beginTransition(mode)
    end, debug.traceback)
    if not ok then visualFailure(failure) end
    updateThemeButtons()
end

local function tickTheme()
    local tr = T.transition
    if tr then
        local alpha = math.clamp((os.clock() - tr.started) / tr.duration, 0, 1)
        local eased = 0.5 - math.cos(alpha * math.pi) * 0.5
        tr.p = tr.from + (tr.to - tr.from) * eased
        drawTransition(tr)
        if alpha >= 1 then settleTransition() end
    elseif THEME_SETTINGS.Enabled and T.automatic
        and not S.closing and os.clock() >= T.deadline then
        beginTransition(T.current == "MIDNIGHT" and "IVORY" or "MIDNIGHT")
    end
end

fit = function()
    if not S.alive then return end
    local size = bounds.AbsoluteSize
    if size.X <= 0 or size.Y <= 0 then return end

    local availableWidth = math.max(1, size.X - 20)
    local availableHeight = math.max(1, size.Y - 20)
    local scale = math.min(1, availableWidth / 280, availableHeight / 205)
    scale = math.max(0.01, scale)

    local width = math.min(460, availableWidth / scale)
    local totalHeight = S.collapsed and 152
        or math.min(846, availableHeight / scale)
    local height = math.max(1, totalHeight - 56)

    local dimensionsChanged = width ~= T.width or height ~= T.height or scale ~= T.scale
    if dimensionsChanged and T.transition then
        -- A viewport resize commits the currently requested destination.
        -- No stale-size masks survive into the new layout.
        settleTransition()
    end

    T.width, T.height, T.scale = width, height, scale
    shellScale.Scale = scale
    shell.Size = UDim2.fromOffset(width, totalHeight)
    shell.Position = UDim2.fromOffset(
        math.clamp(shell.Position.X.Offset, 0, math.max(0, size.X - width * scale)),
        math.clamp(shell.Position.Y.Offset, 0,
            math.max(0, size.Y - totalHeight * scale - 7))
    )

    for _, v in pairs(views) do
        v.body.Visible = not S.collapsed
        v.divider.Visible = not S.collapsed
        v.footer.Visible = not S.collapsed
        v.collapse.Text = S.collapsed and "+" or "-"
        v.reflow(width)
    end
    syncLighting()
end

cleanup = function(destroyGui)
    if not S.alive then return end
    S.alive, S.auto, S.targetAuto = false, false, false
    T.automatic = false
    dragInput = nil

    if S.outstanding then
        warn("[TDX] Removed with an unresolved request. Removal does not cancel server processing.")
    end

    for _, connection in ipairs(S.connections) do
        connection:Disconnect()
    end
    table.clear(S.connections)

    for key, tween in pairs(visualTweens) do
        visualTweens[key] = nil
        tween:Cancel()
        tween:Destroy()
    end

    local tr = T.transition
    T.transition = nil
    table.clear(T.mirrors)
    if tr and tr.layer then tr.layer:Destroy() end

    table.clear(feedbackRecords)
    table.clear(S.seen)
    table.clear(S.fresh)
    table.clear(S.targeted)
    S.job, S.uncertain = nil, nil
    table.clear(views)
    table.clear(themeButtons)
    if destroyGui then gui:Destroy() end
end

requestClose = function(reason)
    if not S.alive then return end
    S.auto, S.targetAuto = false, false
    T.automatic = false
    table.clear(S.fresh)
    dragInput = nil

    outstanding()
    if S.outstanding then
        S.closing = true
        if S.job then
            S.job.cancelReason = reason or "Close requested."
        end
        message(
            S.uncertain
                and "Close queued. Outcome unknown; waiting for a level update."
                or "Close queued. Waiting for the outstanding purchase.",
            S.uncertain ~= nil
        )
        updateThemeButtons()
    else
        cleanup(true)
    end
end

dispatch = function(command, argument)
    if not S.alive then return end

    if command == "close" then
        requestClose("Close requested.")
        return
    end

    if T.transition then return end

    if command == "collapse" then
        dragInput = nil
        resetFeedback()
        S.collapsed = not S.collapsed
        fit()
        return
    end

    if command == "cancel" then
        if S.job then
            S.job.cancelReason = S.job.cancelReason or "Canceled."
            message("Stopping after any outstanding purchase settles.")
            guarded(tickJob)
        elseif S.uncertain then
            message("Outcome unknown. Cancel cannot undo a dispatched request.", true)
        elseif not S.closing then
            message("No active upgrade job.")
        end
        return
    end

    guarded(function()
        if S.closing then return end

        if command == "path" then
            if S.job or S.uncertain then return end
            if type(argument) ~= "number" or not PATHS[argument] then return end
            S.path = argument
            message(PATHS[argument].name
                .. " selected. Secondary path upgrades first.")

        elseif command == "start" then
            start(false)

        elseif command == "auto" then
            if S.uncertain then
                message("Resolve the previous outcome before enabling Auto.", true)
                return
            end

            S.auto = not S.auto
            table.clear(S.fresh)
            table.clear(S.seen)
            scanElapsed = 0

            if S.auto then
                for _, tower in pairs(API.Towers.GetTowers()) do
                    S.seen[tower] = true
                end
                local tower = current()
                if tower then S.seen[tower] = true end
            end

            message(S.auto
                and "Auto: select a newly observed owned tower within 12 seconds."
                or "Auto OFF. Existing job unchanged; Cancel stops it.")

        elseif command == "target" then
            if S.uncertain then
                message("Resolve the previous outcome before enabling targeting.", true)
                return
            end
            S.targetAuto = not S.targetAuto
            message(S.targetAuto
                and "Targeting: cycles C once per selected owned tower."
                or "Automatic targeting disabled.")
        end
    end)
end

local uiElapsed = 0.15
local uncertainReadFailed = false
local presentationFailed = false

local function renderSafely()
    if not S.alive or presentationFailed then return end
    local ok, failure = xpcall(render, debug.traceback)
    if ok then return end

    -- Discard any broken mirrors before trying the persistent views.
    visualFailure(failure)
    local recovered, renderError = xpcall(render, debug.traceback)
    if not recovered then
        presentationFailed = true
        S.auto, S.targetAuto = false, false
        table.clear(S.fresh)
        if S.job then
            S.job.cancelReason = "Presentation stopped."
        end
        warn("[TDX] Presentation stopped: " .. tostring(renderError))
        requestClose("Presentation stopped.")
    end
end

-- One heartbeat owns gameplay progression, display sampling, and the reveal.
-- Theme animation never starts another gameplay worker.
connect(RunService.Heartbeat, function(dt)
    if not S.alive then return end

    if not S.faulted then
        guarded(function()
            if S.uncertain then
                tickUncertain()
            elseif S.job then
                tickJob()
            end

            if S.closing then return end

            tickAuto(dt)

            local ok, failure = pcall(tickTargeting)
            if not ok then
                S.targetAuto = false
                message("Automatic targeting stopped; see console.", true)
                warn("[TDX] " .. tostring(failure))
            end
        end)
    elseif S.uncertain and not uncertainReadFailed and not S.busy then
        -- A fault never authorizes another purchase. Only reconcile the
        -- request already dispatched, using the existing read-only check.
        S.busy = true
        local ok, failure = pcall(tickUncertain)
        S.busy = false
        if not ok then
            uncertainReadFailed = true
            message("Client error; outcome unknown. No retry issued.", true)
            warn("[TDX] " .. tostring(failure))
        end
    end

    if not S.alive then return end
    if S.closing and not S.outstanding and not S.busy then
        cleanup(true)
        return
    end

    uiElapsed = uiElapsed + dt
    if uiElapsed >= 0.15 then
        uiElapsed = 0
        if not S.faulted then
            guarded(sampleDisplay)
        end
        renderSafely()
    end

    if not S.alive then return end

    local ok, failure = xpcall(function()
        tickTheme()

        local tr = T.transition
        if tr and tr.beam then
            -- Feather the light at either endpoint, including reversals.
            -- The slice masks remain opaque: this fades only the beam,
            -- never the outgoing or incoming presentation.
            local edge = math.min(tr.p, 1 - tr.p)
            tr.beam.BackgroundTransparency =
                1 - math.clamp(edge / 0.065, 0, 1)
        end
    end, debug.traceback)

    if not ok then
        visualFailure(failure)
    end
end)

-- Initialize both existing presentations before permitting automatic changes.
-- Their controls already share dispatch; no additional control connections
-- or view construction is needed here.
destroyTransition()
fit()
resetThemeDeadline()
updateThemeButtons()
guarded(sampleDisplay)
renderSafely()

if S.alive then
    syncLighting()
    if not S.faulted then
        print("[TDX] Midnight + Ivory loaded. One shared gameplay worker; "
            .. "server/client purchase limits still apply.")
    end
end
