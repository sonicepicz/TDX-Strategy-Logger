-- TDX / MIDNIGHT + IVORY
-- One application, one gameplay worker, two independently constructed views.
-- Uses the exact TDX client contracts supplied in the original scripts.
-- Reviewed statically; not runtime-tested inside Roblox/TDX.
--
-- The reveal uses temporary, pruned horizontal GUI slices. Each slice clips
-- an incoming-view copy at a calculated diagonal boundary. These copies have
-- no callbacks, workers, or event connections. Shared display updates are
-- mirrored into them while the transition runs.
--
-- Manual theme requests can reverse an in-progress reveal without resetting
-- gameplay, rebuilding the views, or starting another worker.

local function positive(value, fallback, minimum)
    if type(value) ~= "number" or value ~= value or math.abs(value) == math.huge then
        return fallback
    end
    return math.max(minimum, value)
end

local function make(className, parent, properties)
    local object = Instance.new(className)
    for key, value in pairs(properties or {}) do
        object[key] = value
    end
    object.Parent = parent
    return object
end

local function corner(object, radius)
    return make("UICorner", object, {
        CornerRadius = UDim.new(0, radius or 12),
    })
end

local function stroke(object, color, transparency, thickness)
    return make("UIStroke", object, {
        Color = color,
        Transparency = transparency or 0,
        Thickness = thickness or 1,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    })
end

local function gradient(object, first, last, rotation)
    return make("UIGradient", object, {
        Color = ColorSequence.new(first, last),
        Rotation = rotation or 0,
    })
end

local function set(object, property, value)
    if object[property] ~= value then
        object[property] = value
    end
    local copies = T.mirrors[object]
    if copies then
        for _, copy in ipairs(copies) do
            if copy[property] ~= value then
                copy[property] = value
            end
        end
    end
end

local function replaceTween(key, object, info, goals)
    local old = visualTweens[key]
    if old then
        visualTweens[key] = nil
        old:Cancel()
        old:Destroy()
    end
    local tween = TweenService:Create(object, info, goals)
    visualTweens[key] = tween
    tween:Play()
    return tween
end

local function rule(parent, position, size, color, transparency)
    local object = make("Frame", parent, {
        Position = position,
        Size = size,
        BackgroundColor3 = color,
        BackgroundTransparency = transparency or 0,
        BorderSizePixel = 0,
        Active = false,
        ZIndex = 2,
    })

    local function isLineThickness(dimension)
        return dimension.Scale == 0
            and dimension.Offset > 0

        object:SetAttribute("TransitionSkip", true)

-- Ivory decorations share one opacity envelope.
-- Existing Heartbeat applies it; existing cleanup owns every tween.
local function ivoryState(v)
    if not v.ivory then
        v.ivory = {
            gain = 0,
            surfaces = {},
        }
    end
    return v.ivory
end

local function ivoryMake(className, parent, properties)
    local object = make(className, parent, properties)
    object:SetAttribute("TransitionSkip", true)
    return object
end

local function ivoryRound(object, radius)
    local rounding = corner(object, radius)
    rounding:SetAttribute("TransitionSkip", true)
    return rounding
end

local function ivoryValue(v, name, initial)
    return ivoryMake("NumberValue", v.root, {
        Name = name,
        Value = initial,
    })
end

local function ivoryTrack(v, object, property, alpha, weight, pulse)
    local state = ivoryState(v)
    local name = object.Name

    -- Stronger everywhere, with extra emphasis on subtle material details.
    local boost = 2.8

    if name:find("PearlUnderlight", 1, true) then
        boost = 5
    elseif name == "DustHalo" then
        boost = 5
    elseif name == "IvoryInnerGlass" then
        boost = 2.1
    elseif name == "IvoryHoverReflection" then
        boost = 3.2
    end

    local opacity = math.clamp((1 - alpha) * boost, 0, 0.96)

    state.surfaces[#state.surfaces + 1] = {
        object = object,
        property = property,
        alpha = 1 - opacity,
        weight = weight,
        pulse = pulse,
    }

    object[property] = 1
    return object
end

local function ivoryFrame(v, parent, properties, weight, pulse)
    local alpha = properties.BackgroundTransparency
    if alpha == nil then alpha = 0 end

    properties.BackgroundTransparency = 1
    properties.BorderSizePixel = 0
    properties.Active = false
    properties.Interactable = false
    properties.Selectable = false

    local object = ivoryMake("Frame", parent, properties)
    if alpha < 1 then
        ivoryTrack(
            v, object, "BackgroundTransparency",
            alpha, weight, pulse
        )
    end
    return object
end

local function ivoryStroke(v, parent, color, alpha, thickness, weight, pulse)
    local outline = stroke(parent, color, 1, thickness)
    outline:SetAttribute("TransitionSkip", true)
    ivoryTrack(v, outline, "Transparency", alpha, weight, pulse)
    return outline
end

local function ivoryAnimate(v, object, duration, delayTime, reverses, goals)
    local name = object.Name
    local sparkle = name:find("Sparkle", 1, true) ~= nil

    local speed = sparkle and 0.75 or 0.55
    local durationFloor = sparkle and 1.2 or 2.2
    local visibleDelay = math.min((delayTime or 0) * 0.16, 2.5)

    -- Sparkles should already have a visible presence at first display.
    if object:IsA("NumberValue") and sparkle then
        object.Value = math.max(object.Value, 0.30)
    end

    local tween = TweenService:Create(
        object,
        TweenInfo.new(
            math.max(durationFloor, duration * speed),
            Enum.EasingStyle.Sine,
            Enum.EasingDirection.InOut,
            -1,
            reverses,
            visibleDelay
        ),
        goals
    )

    visualTweens[object] = tween
    v.lights[#v.lights + 1] = tween
    return tween
end

local function ivoryShimmer(v, parent, name, duration, delayTime, alpha, weight)
    local _, reflection = ivoryBand(
        v, parent, name, alpha or 0.65, 24, weight
    )

    -- Both endpoints are outside the panel.
    -- The repeat reset therefore happens while the band is invisible.
    ivoryAnimate(v, reflection, duration, delayTime, false, {
        Offset = Vector2.new(1.5, 0),
    })
end

local function tickIvoryPresentation(dt)
    local v = views.IVORY
    if not v or not v.ivory then return end

    local state = v.ivory
    local target = 0
    local tr = T.transition

    if not S.collapsed then
        if tr then
            if tr.outgoing == "IVORY" or tr.incoming == "IVORY" then
                target = 1
            end
        elseif T.current == "IVORY" then
            target = 1
        end
    end

    -- Faster withdrawal; slower, sunrise-like arrival.
    local response = target > state.gain and 5.5 or 9
    local blend = 1 - math.exp(-math.min(dt, 0.1) * response)
    state.gain = state.gain + (target - state.gain) * blend

    if math.abs(target - state.gain) < 0.001 then
        state.gain = target
    end

    for _, entry in ipairs(state.surfaces) do
        local weight = entry.weight and entry.weight.Value or 1
        local pulse = entry.pulse and entry.pulse.Value or 1
        local opacity = (1 - entry.alpha) * state.gain * weight * pulse
        local alpha = 1 - math.clamp(opacity, 0, 1)

        if math.abs(entry.object[entry.property] - alpha) > 0.0005 then
            entry.object[entry.property] = alpha
        end
    end
end

local function button(v, parent, value, position, size, command, argument)
    local object = make("TextButton", parent, {
        Position = position,
        Size = size,
        BackgroundColor3 = v.C.surface,
        BorderSizePixel = 0,
        Text = value,
        TextColor3 = v.C.text,
        TextSize = 12,
        Font = v.name == "MIDNIGHT" and Enum.Font.GothamBold or Enum.Font.GothamMedium,
        TextWrapped = true,
        AutoButtonColor = false,
        ZIndex = 4,
    })
    corner(object, v.name == "MIDNIGHT" and 9 or 12)

    -- Keep primary actions solid; use each theme's secondary glass finish.
    if v.name == "MIDNIGHT" and command ~= "start" then
        object.BackgroundTransparency = 0.32
    elseif v.name == "IVORY" and command ~= "start" then
        object.BackgroundTransparency = 0.44
    end

    local outline = stroke(object, v.C.border, 0.25)
    local feedback = stroke(object, v.C.secondary, 1, 2)
    local scale = make("UIScale", object, {Scale = 1})
    local record = {
        object = object,
        scale = scale,
        outline = feedback,
        hovered = false,
    }
    feedbackRecords[#feedbackRecords + 1] = record

    if v.name == "IVORY" and command ~= "start" then
        local hover = ivoryValue(v, "IvoryButtonLight", 0)
        record.ivoryHover = hover

        local wash = ivoryFrame(v, object, {
            Name = "IvoryHoverReflection",
            Position = UDim2.fromOffset(1, 1),
            Size = UDim2.new(1, -2, 1, -2),
            BackgroundColor3 = Color3.new(1, 1, 1),
            BackgroundTransparency = 0.72,
            ZIndex = 1,
        }, hover)
        ivoryRound(wash, 11)
        ivoryMake("UIGradient", wash, {
            Rotation = 22,
            Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0, 0.80),
                NumberSequenceKeypoint.new(0.38, 0.20),
                NumberSequenceKeypoint.new(0.68, 0.55),
                NumberSequenceKeypoint.new(1, 1),
            }),
        })

        local lip = ivoryFrame(v, object, {
            Name = "IvoryButtonLip",
            Position = UDim2.fromOffset(12, 1),
            Size = UDim2.new(1, -24, 0, 1),
            BackgroundColor3 = Color3.new(1, 1, 1),
            BackgroundTransparency = 0.39,
            ZIndex = 2,
        })
        ivoryMake("UIGradient", lip, {
            Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0, 1),
                NumberSequenceKeypoint.new(0.35, 0),
                NumberSequenceKeypoint.new(0.70, 0.30),
                NumberSequenceKeypoint.new(1, 1),
            }),
        })
    end

    local function animate(pressed)
        if not S.alive then return end
        replaceTween(scale, scale, TweenInfo.new(0.14), {
            Scale = pressed and 0.982 or 1,
        })
        local ivory = v.name == "IVORY"

        replaceTween(feedback, feedback, TweenInfo.new(0.16), {
            Transparency = pressed and (ivory and 0.48 or 0.1)
                or (record.hovered and (ivory and 0.68 or 0.35) or 1),
        })

        if record.ivoryHover then
            replaceTween(
                record.ivoryHover,
                record.ivoryHover,
                TweenInfo.new(
                    pressed and 0.12 or 0.28,
                    Enum.EasingStyle.Sine,
                    Enum.EasingDirection.Out
                ),
                {Value = pressed and 0.62 or (record.hovered and 1 or 0)}
            )
        end
    end

    connect(object.MouseEnter, function()
        record.hovered = true
        animate(false)
    end)
    connect(object.MouseLeave, function()
        record.hovered = false
        animate(false)
    end)
    connect(object.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            animate(true)
        end
    end)
    connect(object.InputEnded, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            animate(false)
        end
    end)
    connect(object.Activated, function()
        if dispatch then dispatch(command, argument) end
    end)
    return object, outline
end

local function resetFeedback()
    for _, record in ipairs(feedbackRecords) do
        for _, key in ipairs({record.scale, record.outline}) do
            local tween = visualTweens[key]
            if tween then
                visualTweens[key] = nil
                tween:Cancel()
                tween:Destroy()
            end
        end
        record.hovered = false
        record.scale.Scale = 1
        record.outline.Transparency = 1

        if record.ivoryHover then
            local tween = visualTweens[record.ivoryHover]
            if tween then
                visualTweens[record.ivoryHover] = nil
                tween:Cancel()
                tween:Destroy()
            end
            record.ivoryHover.Value = 0
        end
    end
end

local function addMidnightAurora(v)
    if v.name ~= "MIDNIGHT" then return end

    local atmosphere = make("Frame", v.root, {
        Name = "MidnightAurora",
        Size = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        ClipsDescendants = true,
        Active = false,
        Interactable = false,
        Selectable = false,
        ZIndex = 1,
    })

    -- Decorative instances stay out of the temporary reveal copies.
    atmosphere:SetAttribute("TransitionSkip", true)

    local function animate(object, duration, delayTime, goals)
        local tween = TweenService:Create(
            object,
            TweenInfo.new(
                duration,
                Enum.EasingStyle.Sine,
                Enum.EasingDirection.InOut,
                -1,
                true,
                delayTime or 0
            ),
            goals
        )

        visualTweens[object] = tween
        v.lights[#v.lights + 1] = tween
    end

    -- Bright overlapping curtains. Gradient rotation gives diagonal
    -- shading without rotating Frames or breaking ancestor clipping.
    local ribbons = {
        {
            x = 0.08, width = 0.72, drift = 0.22,
            rotation = -18, duration = 14, delay = 0,
            color = Color3.fromRGB(35, 110, 255),
            alpha = 0.42, peak = 0.12,
        },
        {
            x = 0.35, width = 0.58, drift = -0.18,
            rotation = 16, duration = 19, delay = 0.5,
            color = Color3.fromRGB(125, 65, 255),
            alpha = 0.48, peak = 0.18,
        },
        {
            x = 0.66, width = 0.64, drift = 0.20,
            rotation = -12, duration = 23, delay = 1,
            color = Color3.fromRGB(30, 170, 255),
            alpha = 0.52, peak = 0.20,
        },
        {
            x = 0.94, width = 0.55, drift = -0.25,
            rotation = 22, duration = 17, delay = 0.3,
            color = Color3.fromRGB(155, 85, 255),
            alpha = 0.46, peak = 0.14,
        },
    }

    for index, spec in ipairs(ribbons) do
        local ribbon = make("Frame", atmosphere, {
            Name = "AuroraRibbon" .. index,
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromScale(spec.x, 0.5),
            Size = UDim2.fromScale(spec.width, 1.8),
            BackgroundColor3 = spec.color,
            BackgroundTransparency = spec.alpha,
            BorderSizePixel = 0,
            Active = false,
            Interactable = false,
            Selectable = false,
            ZIndex = 1,
        })

        make("UIGradient", ribbon, {
            Rotation = spec.rotation,
            Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0, 1),
                NumberSequenceKeypoint.new(0.14, 0.94),
                NumberSequenceKeypoint.new(0.30, 0.58),
                NumberSequenceKeypoint.new(0.44, 0.12),
                NumberSequenceKeypoint.new(0.52, 0),
                NumberSequenceKeypoint.new(0.63, 0.30),
                NumberSequenceKeypoint.new(0.82, 0.88),
                NumberSequenceKeypoint.new(1, 1),
            }),
        })

        animate(ribbon, spec.duration, spec.delay, {
            Position = UDim2.fromScale(spec.x + spec.drift, 0.5),
            BackgroundTransparency = spec.peak,
        })
    end

    -- Sparse, clearly visible points. Several sit in the side gutters
    -- so opaque UI details cannot conceal the entire particle field.
    local particles = {
        {x = 0.025, y = 0.18, dx = 0.009, dy = -0.045, size = 4, duration = 7, delay = 0},
        {x = 0.970, y = 0.27, dx = -0.008, dy = 0.040, size = 3, duration = 9, delay = 0.4},
        {x = 0.030, y = 0.48, dx = 0.006, dy = -0.050, size = 3, duration = 10, delay = 1.1},
        {x = 0.970, y = 0.67, dx = -0.009, dy = -0.045, size = 4, duration = 8, delay = 0.7},
        {x = 0.035, y = 0.88, dx = -0.006, dy = -0.035, size = 3, duration = 12, delay = 1.5},
        {x = 0.180, y = 0.12, dx = 0.025, dy = 0.035, size = 4, duration = 8, delay = 0.2},
        {x = 0.420, y = 0.31, dx = -0.025, dy = -0.055, size = 3, duration = 11, delay = 0.9},
        {x = 0.730, y = 0.44, dx = 0.020, dy = -0.040, size = 4, duration = 9, delay = 1.3},
        {x = 0.270, y = 0.65, dx = 0.030, dy = -0.050, size = 3, duration = 13, delay = 0.5},
        {x = 0.620, y = 0.86, dx = -0.025, dy = -0.045, size = 4, duration = 10, delay = 1.8},
    }

    for index, spec in ipairs(particles) do
        local color = index % 2 == 0
            and Color3.fromRGB(225, 242, 255)
            or Color3.fromRGB(110, 190, 255)

        local point = make("Frame", atmosphere, {
            Name = "AuroraPoint" .. index,
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromScale(spec.x, spec.y),
            Size = UDim2.fromOffset(spec.size, spec.size),
            BackgroundColor3 = color,
            BackgroundTransparency = 0.65,
            BorderSizePixel = 0,
            Active = false,
            Interactable = false,
            Selectable = false,
            ZIndex = 2,
        })
        corner(point, 8)

        -- Two inexpensive circular halos around each point.
        local outer = make("Frame", point, {
            Name = "OuterGlow",
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromScale(0.5, 0.5),
            Size = UDim2.fromOffset(spec.size + 5, spec.size + 5),
            BackgroundColor3 = color,
            BackgroundTransparency = 0.97,
            BorderSizePixel = 0,
            Active = false,
            Interactable = false,
            Selectable = false,
            ZIndex = 1,
        })
        corner(outer, 24)

        local inner = make("Frame", point, {
            Name = "InnerGlow",
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromScale(0.5, 0.5),
            Size = UDim2.fromOffset(spec.size + 2.5, spec.size + 2.5),
            BackgroundColor3 = color,
            BackgroundTransparency = 0.90,
            BorderSizePixel = 0,
            Active = false,
            Interactable = false,
            Selectable = false,
            ZIndex = 1,
        })
        corner(inner, 16)

        -- Forward: drift and brighten. Reverse: return and fade.
        animate(point, spec.duration, spec.delay, {
            Position = UDim2.fromScale(
                spec.x + spec.dx,
                spec.y + spec.dy
            ),
            BackgroundTransparency = 0.02,
        })
        animate(inner, spec.duration, spec.delay, {
            BackgroundTransparency = 0.48,
        })
        animate(outer, spec.duration, spec.delay, {
            BackgroundTransparency = 0.82,
        })
    end
end

local function addIvoryAtmosphere(v)
    if v.name ~= "IVORY" then return end
    if v.root:FindFirstChild("IvoryAtmosphere") then return end

    ivoryState(v)

    local white = Color3.new(1, 1, 1)
    local warmWhite = Color3.fromRGB(255, 247, 224)
    local paleGold = Color3.fromRGB(228, 192, 120)
    local champagne = Color3.fromRGB(185, 139, 62)

    local atmosphere = ivoryFrame(v, v.root, {
        Name = "IvoryAtmosphere",
        Size = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1,
        ClipsDescendants = true,
        ZIndex = 1,
    })
    ivoryRound(atmosphere, 24)

    -- Unrotated full-window frames preserve ancestor clipping.
    -- Only gradient coordinates rotate.
    local ribbons = {
        {angle = 28, alpha = 0.65, time = 21, delay = 0.0,
            start = Vector2.new(-0.28, -0.10),
            finish = Vector2.new(0.22, 0.08),
            edge = paleGold, center = warmWhite},

        {angle = -24, alpha = 0.72, time = 29, delay = 1.3,
            start = Vector2.new(0.33, 0.09),
            finish = Vector2.new(-0.24, -0.06),
            edge = champagne, center = warmWhite},

        {angle = 39,  alpha = 0.48, time = 25, delay = 0.4,
            start = Vector2.new(0.05, -0.20),
            finish = Vector2.new(-0.18, 0.15),
            edge = warmWhite, center = white},

        {angle = -31, alpha = 0.80, time = 34, delay = 2.1,
            start = Vector2.new(-0.31, 0.13),
            finish = Vector2.new(0.27, -0.10),
            edge = paleGold, center = white},
    }

    for index, spec in ipairs(ribbons) do
        local ribbon = ivoryFrame(v, atmosphere, {
            Name = "ChampagneRibbon" .. index,
            Size = UDim2.fromScale(1, 1),
            BackgroundColor3 = white,
            BackgroundTransparency = spec.alpha,
            ZIndex = 1,
        })

        local reflection = ivoryMake("UIGradient", ribbon, {
            Rotation = spec.angle,
            Offset = spec.start,
            Color = ColorSequence.new({
                ColorSequenceKeypoint.new(0, spec.edge),
                ColorSequenceKeypoint.new(0.35, spec.edge),
                ColorSequenceKeypoint.new(0.55, spec.center),
                ColorSequenceKeypoint.new(1, white),
            }),
            Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0.00, 1.00),
                NumberSequenceKeypoint.new(0.12, 0.98),
                NumberSequenceKeypoint.new(0.28, 0.72),
                NumberSequenceKeypoint.new(0.43, 0.18),
                NumberSequenceKeypoint.new(0.53, 0.04),
                NumberSequenceKeypoint.new(0.65, 0.36),
                NumberSequenceKeypoint.new(0.84, 0.94),
                NumberSequenceKeypoint.new(1.00, 1.00),
            }),
        })

        ivoryAnimate(v, reflection, spec.time, spec.delay, true, {
            Offset = spec.finish,
        })
    end

    -- Broad horizon reflection: depth without a gold wash.
    local horizon = ivoryFrame(v, atmosphere, {
        Name = "IvoryHorizon",
        Position = UDim2.fromScale(0, 0.65),
        Size = UDim2.fromScale(1, 0.35),
        BackgroundColor3 = warmWhite,
        BackgroundTransparency = 0.38,
        ZIndex = 1,
    })
    ivoryMake("UIGradient", horizon, {
        Rotation = 90,
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 1),
            NumberSequenceKeypoint.new(0.60, 0.60),
            NumberSequenceKeypoint.new(1, 0.08),
        }),
    })

    -- A restrained side falloff gives the white center more presence.
    local falloff = ivoryFrame(v, atmosphere, {
        Name = "PorcelainEdgeFalloff",
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = Color3.fromRGB(188, 181, 168),
        BackgroundTransparency = 0.91,
        ZIndex = 1,
    })
    ivoryMake("UIGradient", falloff, {
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 0.15),
            NumberSequenceKeypoint.new(0.13, 0.92),
            NumberSequenceKeypoint.new(0.35, 1),
            NumberSequenceKeypoint.new(0.75, 1),
            NumberSequenceKeypoint.new(1, 0.30),
        }),
    })

    -- Main-window reflection. There is a real invisible rest interval.
    ivoryShimmer(v, atmosphere, "WindowPearlReflection", 18, 9, 0.67)

    -- Fourteen deterministic motes. Most occupy exposed gutters/gaps.
    -- The field sits behind content, never over labels.
    local particles = {
        {0.025, 0.15,  0.007, -0.026, 2, 17},
        {0.976, 0.22, -0.006,  0.031, 3, 22},
        {0.028, 0.36,  0.004, -0.038, 2, 19},
        {0.975, 0.45, -0.005, -0.029, 3, 26},
        {0.023, 0.59,  0.006, -0.027, 2, 23},
        {0.979, 0.69, -0.005,  0.024, 2, 18},
        {0.026, 0.81,  0.005, -0.031, 3, 28},
        {0.974, 0.92, -0.006, -0.026, 2, 24},
        {0.29, 0.12,  0.014, -0.010, 2, 21},
        {0.67, 0.13, -0.018,  0.009, 3, 27},
        {0.22, 0.46,  0.018, -0.021, 2, 20},
        {0.78, 0.48, -0.015,  0.026, 4, 31},
        {0.36, 0.76,  0.019, -0.022, 2, 25},
        {0.70, 0.96, -0.017, -0.018, 3, 29},
    }

    for index, spec in ipairs(particles) do
        local breathing = ivoryValue(v, "DustBreath" .. index, 0.65)
        local color = index % 3 == 0 and champagne or paleGold
        local particleSize = spec[5] + 2
        local point = ivoryFrame(v, atmosphere, {
            Name = "IvoryDust" .. index,
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromScale(spec[1], spec[2]),
            Size = UDim2.fromOffset(particleSize, particleSize),
            BackgroundColor3 = color,
            BackgroundTransparency = 0.18,
            ZIndex = 3,
        }, nil, breathing)
        ivoryRound(point, 8)

        -- A very faint outline gives white dust definition on white.
        ivoryStroke(v, point, champagne, 0.40, 1, nil, breathing)

        ivoryAnimate(v, point, spec[6], index * 0.17, true, {
            Position = UDim2.fromScale(
                math.clamp(spec[1] + spec[3] * 1.6, 0.018, 0.982),
                math.clamp(spec[2] + spec[4] * 1.8, 0.025, 0.975)
            ),
        })
        ivoryAnimate(
            v,
            breathing,
            6.5 + index * 0.43,
            index * 0.31,
            true,
            {Value = 1}
        )

        if index % 2 == 0 then
            local halo = ivoryFrame(v, point, {
                Name = "DustHalo",
                AnchorPoint = Vector2.new(0.5, 0.5),
                Position = UDim2.fromScale(0.5, 0.5),
                Size = UDim2.fromOffset(particleSize + 9, particleSize + 9),
                BackgroundColor3 = paleGold,
                BackgroundTransparency = 0.86,
                ZIndex = 1,
            }, nil, breathing)
            ivoryRound(halo, 12)
        end
    end
end

local function addLighting(v)
    -- Ivory now has one atmosphere owner.
    -- Keep Midnight's existing lighting path exactly as before.
    if v.name == "IVORY" then return end

    local dark = v.name == "MIDNIGHT"
    local ambient = make("Frame", v.root, {
        Name = "Atmosphere",
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BackgroundTransparency = dark and 0.70 or 0.12,
        BorderSizePixel = 0,
        Active = false,
        Interactable = false,
        ZIndex = 1,
    })
    corner(ambient, dark and 16 or 24)

    local ambientGradient = make("UIGradient", ambient, {
        Rotation = dark and 32 or 24,
        Offset = Vector2.new(-1.2, -0.12),
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, v.C.secondary),
            ColorSequenceKeypoint.new(0.5,
                dark and Color3.fromRGB(201, 226, 255)
                    or Color3.fromRGB(255, 253, 225)),
            ColorSequenceKeypoint.new(1, v.C.accent),
        }),
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 1),
            NumberSequenceKeypoint.new(0.28, 1),
            NumberSequenceKeypoint.new(0.42, 0.90),
            NumberSequenceKeypoint.new(0.5, dark and 0.20 or 0.70),
            NumberSequenceKeypoint.new(0.58, 0.90),
            NumberSequenceKeypoint.new(0.72, 1),
            NumberSequenceKeypoint.new(1, 1),
        }),
    })

    local tween = TweenService:Create(
        ambientGradient,
        TweenInfo.new(dark and 14 or 13, Enum.EasingStyle.Sine,
            Enum.EasingDirection.InOut, -1, true, 0.5),
        {Offset = Vector2.new(1.2, 0.16)}
    )
    visualTweens[ambientGradient] = tween
    v.lights[#v.lights + 1] = tween

    if dark then
        -- These HUD rails are decorative background elements. Mark them so
        -- the cinematic slice renderer does not clone them once per slice;
        -- doing so would multiply the rails into the visible line pattern
        -- during the reveal. They remain part of the normal Midnight view.
        for i = 1, 4 do
            local rail = rule(v.root, UDim2.new(i / 5, 0, 0, 0),
                UDim2.new(0, 1, 1, 0), v.C.accent, 0.96)
            rail.ZIndex = 1
            rail.Name = "HUDRailV" .. i
            rail:SetAttribute("TransitionSkip", true)
        end
        for i = 1, 5 do
            local rail = rule(v.root, UDim2.new(0, 0, i / 6, 0),
                UDim2.new(1, 0, 0, 1), v.C.accent, 0.975)
            rail.ZIndex = 1
            rail.Name = "HUDRailH" .. i
            rail:SetAttribute("TransitionSkip", true)
        end
    else
        rule(v.root, UDim2.new(1, -30, 0, 103),
            UDim2.new(0, 1, 1, -120), v.C.secondary, 0.86)
    end
end

local function ivorySparkle(v, parent, name, position, duration, delayTime, weight)
    local holder = ivoryFrame(v, parent, {
        Name = name,
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = position,
        Size = UDim2.fromOffset(20, 20),
        BackgroundTransparency = 1,
        ZIndex = 2,
    })

    local breath = ivoryValue(v, name .. "Breath", 0)
    local scale = ivoryMake("UIScale", holder, {Scale = 0.90})

    -- Two soft tapered arms suggest four points without rotating frames.
    for index, size in ipairs({
        UDim2.fromOffset(3, 20),
        UDim2.fromOffset(18, 3),
    }) do
        local arm = ivoryFrame(v, holder, {
            Name = "PearlArm" .. index,
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromScale(0.5, 0.5),
            Size = size,
            BackgroundColor3 = Color3.fromRGB(211, 164, 78),
            BackgroundTransparency = 0.12,
            ZIndex = 1,
        }, weight, breath)
        ivoryRound(arm, 2)
        ivoryMake("UIGradient", arm, {
            Rotation = index == 1 and 90 or 0,

            Color = ColorSequence.new({
                ColorSequenceKeypoint.new(0.00, Color3.fromRGB(205, 151, 62)),
                ColorSequenceKeypoint.new(0.35, Color3.fromRGB(244, 219, 165)),
                ColorSequenceKeypoint.new(0.50, Color3.new(1, 1, 1)),
                ColorSequenceKeypoint.new(0.65, Color3.fromRGB(244, 219, 165)),
                ColorSequenceKeypoint.new(1.00, Color3.fromRGB(205, 151, 62)),
            }),

            Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0.00, 1.00),
                NumberSequenceKeypoint.new(0.20, 0.45),
                NumberSequenceKeypoint.new(0.40, 0.00),
                NumberSequenceKeypoint.new(0.60, 0.00),
                NumberSequenceKeypoint.new(0.80, 0.45),
                NumberSequenceKeypoint.new(1.00, 1.00),

    })

local function message(value, bad)
    if not S.alive then return end
    S.status, S.bad = value, bad == true
end

local function outstanding()
    S.outstanding = S.uncertain ~= nil or (S.job ~= nil and S.job.pending ~= nil)
    gui:SetAttribute("OutstandingUpgrade", S.outstanding)
end

local function current()
    local tower = API.UI.GetCurrentTower()
    if tower and API.Towers.GetTower(tower.Hash) == tower then
        return tower
    end
end

local function cash()
    local wrapper = API.Players.GetLocalPlayerWrapper()
    local value = wrapper and wrapper.Stats and wrapper.Stats:GetCash()
    if value == nil then return 0 end
    assert(type(value) == "number" and value == value
        and math.abs(value) < math.huge, "Invalid client cash.")
    return value
end

local function selectionProblem(tower)
    if not tower then return "Select a tower first." end
    if API.Towers.GetTower(tower.Hash) ~= tower then
        return "Tower no longer exists."
    end
    if not tower.OwnedByLocalPlayer then return "This tower is not yours." end
    if workspace:GetAttribute("FPSMode") or API.Input.GetState() ~= "Normal" then
        return "Finish placement, aiming, or direct control."
    end
    if not tower:Alive() then return "Tower is not alive." end
    if tower:IsRebuilding() then return "Tower is rebuilding." end
    if tower:IsReviving() then return "Tower is reviving." end
end

local function upgradeProblem(tower)
    local why = selectionProblem(tower)
    if why then return why end
    if S.uncertain then
        return "Previous outcome unknown. Waiting for a level update."
    end
    if tower.AbilityHandler and not tower.AbilityHandler:CanUpgrade() then
        return "Cannot upgrade during this ability."
    end
end

local function nextStep(tower, path)
    local lh = tower.LevelHandler
    local data = lh:GetUpgradePathData()
    local a, b = lh:GetLevelOnPath(1), lh:GetLevelOnPath(2)

    if data.OnePathOnly then
        return nil, "This preset needs a two-path tower."
    end
    if a > path.a or b > path.b then
        return nil, "Cannot downgrade to this preset."
    end
    if path.a > #data.Path1Data or path.b > #data.Path2Data then
        return nil, "Preset unavailable."
    end

    local lock = lh:GetPathLockLevel()
    if path.a >= lock and path.b >= lock then
        return nil, "Paths conflict."
    end
    local major = path.a > path.b and 1 or 2
    local lockedPath = lh:GetPath()
    if lockedPath ~= 0 and lockedPath ~= major then
        return nil, "Opposite path locked."
    end
    if a == path.a and b == path.b then
        return nil, nil, true
    end

    local side, before
    if major == 1 then
        if b < path.b then
            side, before = 2, b
        else
            side, before = 1, a
        end
    else
        if a < path.a then
            side, before = 1, a
        else
            side, before = 2, b
        end
    end

    if before + 1 > lh:GetMaxPossibleLevel(side) then
        return nil, "Path locked."
    end

    local cost = lh:GetLevelUpgradeCost(side, 1,
        tower.BuffHandler:GetDiscount(),
        API.Game.GetTowerCostMultiplier(tower.Type),
        tower:GetDynamicPriceScalingData())

    if type(cost) ~= "number" or cost ~= cost or cost < 0 or cost == math.huge then
        return nil, "Price unavailable."
    end
    return {side = side, level = before + 1, cost = cost}
end

local function finish(value, bad)
    S.job = nil
    outstanding()
    message(value, bad)
end

local function tickUncertain()
    local record = S.uncertain
    if not record then return end
    local tower, pending = record.tower, record.pending

    if API.Towers.GetTower(tower.Hash) ~= tower then
        S.uncertain = nil
        outstanding()
        message("Tower removed. Unresolved tower request cleared.")
    elseif tower.LevelHandler:GetLevelOnPath(pending.side) >= pending.level then
        S.uncertain = nil
        outstanding()
        message("Late level update received. Review before restarting.")
    end
end

local function tickJob()
    local job = S.job
    if not job then return end
    local tower = job.tower

    if API.Towers.GetTower(tower.Hash) ~= tower then
        finish("Tower removed; stopped.", true)
        return
    end
    if API.UI.GetCurrentTower() ~= tower then
        job.cancelReason = job.cancelReason or "Selection changed."
    end

    local pending = job.pending
    if pending then
        if tower.LevelHandler:GetLevelOnPath(pending.side) >= pending.level then
            job.pending = nil
            outstanding()
        elseif os.clock() >= pending.deadline then
            S.uncertain = {tower = tower, pending = pending}
            S.auto = false
            table.clear(S.fresh)
            finish("Outcome UNKNOWN. Auto stopped; no retry issued.", true)
            return
        else
            return
        end
    end

    if job.cancelReason then
        finish(job.cancelReason .. " Completed purchases remain.")
        return
    end

    local why = upgradeProblem(tower)
    if why then
        finish(why, true)
        return
    end

    local step, stepError, done = nextStep(tower, job.path)
    if done then
        finish(string.format("Build complete. Reached %d-%d.", job.path.a, job.path.b))
        return
    end
    if not step then
        finish(stepError, true)
        return
    end

    local balance = cash()
    if balance < step.cost then
        message(string.format("Waiting for %d more cash.", math.ceil(step.cost - balance)))
        return
    end
    if API.UI.GetCurrentTower() ~= tower then
        finish("Selection changed.")
        return
    end

    job.pending = {
        side = step.side,
        level = step.level,
        deadline = os.clock() + ACK_TIMEOUT,
    }
    outstanding()
    API.Upgrade:Fire(step.side, false)
    message(string.format("Path %d -> %d. Waiting for level update.", step.side, step.level))
end

local function guarded(callback)
    if not S.alive or S.busy or S.faulted then return end
    S.busy = true
    local ok, failure = xpcall(callback, debug.traceback)
    S.busy = false
    if ok then return end

    if S.job and S.job.pending then
        S.uncertain = {tower = S.job.tower, pending = S.job.pending}
    end
    S.job, S.auto, S.targetAuto, S.faulted = nil, false, false, true
    table.clear(S.fresh)
    outstanding()
    message("Client contract error. Worker stopped; see console.", true)
    warn("[TDX] " .. tostring(failure))
end

local function start(fromAuto)
    if S.job or S.uncertain or S.closing then return end
    local tower = current()
    local why = upgradeProblem(tower)
    if why then
        message(why, true)
        return
    end
    S.fresh[tower] = nil
    S.job = {tower = tower, path = PATHS[S.path]}
    message(fromAuto and "Auto started." or "Started.")
    tickJob()
end

local scanElapsed = 0

local function observeTower(tower, now)
    if not tower or S.seen[tower] then return end
    S.seen[tower] = true
    if S.auto and tower.OwnedByLocalPlayer then
        S.fresh[tower] = now + AUTO_WINDOW
    end
end

local function tickAuto(dt)
    if not S.auto or S.closing or S.uncertain then return end
    local now, tower = os.clock(), current()
    observeTower(tower, now)
    scanElapsed = scanElapsed + dt

    if scanElapsed >= 0.15 then
        scanElapsed = 0
        for _, item in pairs(API.Towers.GetTowers()) do
            observeTower(item, now)
        end
        for item in pairs(S.seen) do
            if API.Towers.GetTower(item.Hash) ~= item then
                S.seen[item], S.fresh[item] = nil, nil
            end
        end
        for item, deadline in pairs(S.fresh) do
            if now > deadline then S.fresh[item] = nil end
        end
    end

    if tower and not S.job then
        local deadline = S.fresh[tower]
        if deadline and now <= deadline and not upgradeProblem(tower) then
            S.fresh[tower] = nil
            start(true)
        end
    end
end

local function tickTargeting()
    if not S.targetAuto or S.closing or S.uncertain or S.outstanding then return end
    local tower = current()
    if not tower or S.targeted[tower] or selectionProblem(tower) then return end
    S.targeted[tower] = true
    API.Target:Fire()
end

local function money(value)
    local result = tostring(math.floor(value))
    local count
    repeat
        result, count = result:gsub("^(-?%d+)(%d%d%d)", "%1,%2")
    until count == 0
    return "$" .. result
end

local function resetThemeDeadline()
    T.deadline = os.clock() + (T.current == "MIDNIGHT"
        and THEME_SETTINGS.MidnightDuration or THEME_SETTINGS.IvoryDuration)

local function syncLighting()
    local tr = T.transition

    for name, v in pairs(views) do
        local active = not S.collapsed and (
            (tr ~= nil and (
                name == tr.outgoing or name == tr.incoming
            ))
            or (tr == nil and name == T.current)
        )

        for _, tween in ipairs(v.lights) do
            -- Avoid repeatedly calling Play on an already running tween.
            if active then
                if tween.PlaybackState ~= Enum.PlaybackState.Playing then
                    tween:Play()
                end
            elseif tween.PlaybackState == Enum.PlaybackState.Playing
                or tween.PlaybackState == Enum.PlaybackState.Delayed then
                tween:Pause()
            end
        end
    end
end

local function destroyTransition()
    local tr = T.transition
    T.transition = nil
    table.clear(T.mirrors)
    if tr and tr.layer then tr.layer:Destroy() end
    for name, v in pairs(views) do
        v.root.Visible = name == T.current
        v.body.ScrollingEnabled = true
    end
end

local function settleTransition()
    local tr = T.transition
    if not tr then return end
    T.current = tr.to == 1 and tr.incoming or tr.outgoing
    destroyTransition()
    resetThemeDeadline()
    updateThemeButtons()
    syncLighting()
end

-- Clone only an object's own properties. Children are visited selectively
-- afterward, so every slice excludes unrelated off-slice UI branches.
local function shallowClone(source)
    local children = source:GetChildren()
    local saved = {}
    for i, item in ipairs(children) do
        saved[i] = item.Archivable
        item.Archivable = false
    end
    local originalArchivable = source.Archivable
    source.Archivable = true
    local ok, copy = pcall(function() return source:Clone() end)
    source.Archivable = originalArchivable
    for i, item in ipairs(children) do
        item.Archivable = saved[i]
    end
    if not ok then error(copy) end
    assert(copy, "Could not create transition slice.")
    return copy, children
end

local function cloneSliceBranch(
    source, parent, top, bottom, clipTop, clipBottom
)
    -- Copies are presentation-only.
    -- NumberValues remain on the original view, where its animation lives.
    if source:IsA("LuaSourceContainer")
        or source:IsA("ValueBase") then
        return nil
    end

    -- The full-panel outline belongs to the persistent view, not
    -- to every clipped copy of that view.
    --
    -- Keep interior card/button strokes and all animated decoration.
    -- Do not globally reject TransitionSkip: the current renderer
    -- intentionally mirrors the atmosphere and material effects.
    if source:IsA("UIStroke") then
        local owner = source.Parent
        if owner == views.MIDNIGHT.root
            or owner == views.IVORY.root then
            return nil
        end
    end

    local isGui = source:IsA("GuiObject")
    local nextTop, nextBottom = clipTop, clipBottom

    if isGui then
        if not source.Visible then
            return nil
        end

        -- If an ancestor has already clipped out this band, nothing
        -- below it can contribute pixels.
        if clipBottom < top or clipTop > bottom then
            return nil
        end

        local clips = source.ClipsDescendants
            or source:IsA("ScrollingFrame")

        if clips then
            local y = source.AbsolutePosition.Y
            local bottomY = y + source.AbsoluteSize.Y

            nextTop = math.max(clipTop, y)
            nextBottom = math.min(clipBottom, bottomY)

            if nextBottom < top or nextTop > bottom then
                return nil
            end
        end

        -- Do not reject a non-clipping branch just because its own
        -- rectangle misses this slice. Its children may extend into it:
        -- aurora ribbons, dust halos, underlights, sparkles, etc.
    end

    local copy, children = shallowClone(source)
    copy.Parent = parent

    local copies = T.mirrors[source]
    if not copies then
        copies = {}
        T.mirrors[source] = copies
    end
    copies[#copies + 1] = copy

    if isGui then
        copy.Active = false
        copy.Interactable = false
        copy.Selectable = false

        if copy:IsA("ScrollingFrame") then
            -- Preserve its copied CanvasPosition, but never let the
            -- passive copy handle scrolling itself.
            copy.ScrollingEnabled = false
        end
    end

    -- Visit visual components recursively too. A UIGradient attached
    -- to a UIStroke, for example, must not lose its mirror mapping.
    for _, item in ipairs(children) do
        cloneSliceBranch(
            item,
            copy,
            top,
            bottom,
            nextTop,
            nextBottom
        )
    end

    return copy
end

local function buildTransitionAnimationMirrors(tr)
    local tracked = {}

    local function track(source, property)
        if not source or not T.mirrors[source] then
            return
        end

        local properties = tracked[source]
        if not properties then
            properties = {}
            tracked[source] = properties
        end

        properties[property] = true
    end

    -- These cover the visual properties animated by the existing
    -- Midnight and Ivory TweenService paths.
    for source in pairs(visualTweens) do
        if typeof(source) == "Instance" then
            if source:IsA("GuiObject") then
                track(source, "Position")
                track(source, "Size")
                track(source, "Rotation")
                track(source, "BackgroundColor3")
                track(source, "BackgroundTransparency")

            elseif source:IsA("UIGradient") then
                track(source, "Offset")
                track(source, "Rotation")
                track(source, "Color")
                track(source, "Transparency")

            elseif source:IsA("UIScale") then
                track(source, "Scale")

            elseif source:IsA("UIStroke") then
                track(source, "Color")
                track(source, "Transparency")
                track(source, "Thickness")
            end
        end
    end

    -- Ivory's material system writes these properties directly,
    -- rather than tweening the surface instances themselves.
    local ivory = views.IVORY
    if ivory and ivory.ivory then
        for _, entry in ipairs(ivory.ivory.surfaces) do
            track(entry.object, entry.property)
        end
    end

    local records = {}

    for source, properties in pairs(tracked) do
        local copies = T.mirrors[source]

        for property in pairs(properties) do
            records[#records + 1] = {
                source = source,
                property = property,
                copies = copies,
                initialized = false,
                last = nil,
            }
        end
    end

    tr.animationMirrors = records
end

local function syncTransitionAnimations(tr)
    if not tr or not tr.animationMirrors then
        return
    end

    for _, record in ipairs(tr.animationMirrors) do
        local value = record.source[record.property]

        if not record.initialized or value ~= record.last then
            record.initialized = true
            record.last = value

            for _, copy in ipairs(record.copies) do
                if copy[record.property] ~= value then
                    copy[record.property] = value
                end
            end
        end
    end
end

local function drawTransition(tr)
    local width, height = T.width, T.height
    local skew = math.min(110, width * 0.24)
    local margin = 52
    local center = -skew / 2 - margin
        + tr.p * (width + skew + margin * 2)

    for _, slice in ipairs(tr.slices) do
        local boundary = center + skew * (slice.middle / height - 0.5)
        slice.mask.Size = UDim2.fromOffset(
            math.clamp(boundary, 0, width), slice.height)
    end

    -- This diagonal beam covers only the small stair-step approximation at
    -- the mask edge. The mask, not beam transparency, reveals the new view.
    tr.beam.Position = UDim2.fromOffset(center, height / 2)
    tr.beam.Rotation = -math.deg(math.atan(skew / height))
end

local function beginTransition(destination)
    if destination == T.current or T.transition or not S.alive then return end
    resetFeedback()
    dragInput = nil

    local tr = {
        outgoing = T.current,
        incoming = destination,
        p = 0,
        from = 0,
        to = 1,
        started = os.clock(),
        duration = THEME_SETTINGS.TransitionDuration,
        slices = {},
    }
    T.transition = tr
    table.clear(T.mirrors)

    -- The mask reveals Ivory's finished material state.
    -- Do not begin with invisible decorations and wait for them to
    -- fade in after the beam has already passed.
    local ivory = views.IVORY
    if ivory and ivory.ivory and not S.collapsed then
        ivory.ivory.gain = 1
        tickIvoryPresentation(0)
    end

    syncLighting()

    tr.layer = make("Frame", viewHost, {
        Name = "CinematicReveal",
        Size = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        ClipsDescendants = true,
        Active = true,
        ZIndex = 20,
    })

    local incoming = views[destination]
    local outgoing = views[T.current]
    incoming.root.Visible = true
    incoming.root.ZIndex = 1
    outgoing.root.Visible = true
    outgoing.root.ZIndex = 2

    for _, v in pairs(views) do v.body.ScrollingEnabled = false end

    local originY = incoming.root.AbsolutePosition.Y

    -- Measure the actual logical-to-screen conversion.
    -- This includes scaling inherited from ancestors, rather than
    -- assuming that shellScale is the only scale in the hierarchy.
    local measuredHeight = incoming.root.AbsoluteSize.Y
    local screenScale = measuredHeight / math.max(T.height, 0.001)

    assert(
        screenScale == screenScale
            and screenScale > 0
            and screenScale < math.huge,
        "Incoming view has no measurable transition scale."
    )

    local physicalHeight = T.height * screenScale
    local desiredCount = math.clamp(
        math.ceil(T.height / 12),
        8,
        72
    )

    -- Extremely small rendered panels must not create more bands
    -- than they have physical pixel rows.
    local count = math.max(
        1,
        math.min(
            desiredCount,
            math.floor(physicalHeight)
        )
    )

    -- Small intentional overlap prevents fractional clipping edges
    -- from exposing the outgoing theme between horizontal bands.
    --
    -- Each copy contains the same opaque root background and uses
    -- the same content origin, so the overlap replaces pixels;
    -- it does not brighten translucent decoration underneath.
    local seamOverlap = 1 / screenScale

    local function sliceEdge(index)
        if index <= 0 then
            return 0
        elseif index >= count then
            return T.height
        end

        local logicalY = index * T.height / count

        -- Logical -> screen: MULTIPLY by scale.
        local screenY = originY + logicalY * screenScale
        local snappedScreenY = math.floor(screenY + 0.5)

        -- Screen -> logical: DIVIDE by scale.
        return math.clamp(
            (snappedScreenY - originY) / screenScale,
            0,
            T.height
        )
    end

    for index = 1, count do
        local y0 = sliceEdge(index - 1)
        local y1 = sliceEdge(index)

        -- Skip any degenerate interval caused by pixel snapping.
        if y1 > y0 then
            local coveredBottom = math.min(
                T.height,
                y1 + seamOverlap
            )
            local maskHeight = coveredBottom - y0

            local mask = make("Frame", tr.layer, {
                Name = "RevealSlice",
                Position = UDim2.fromOffset(0, y0),
                Size = UDim2.fromOffset(0, maskHeight),
                BackgroundTransparency = 1,
                BorderSizePixel = 0,
                ClipsDescendants = true,
                Active = false,
                Interactable = false,
                Selectable = false,

                -- Lower bands deterministically cover the small
                -- overlap from the band immediately above them.
                ZIndex = index,
            })

            local physicalTop = originY + y0 * screenScale
            local physicalBottom =
                originY + coveredBottom * screenScale

            local copy = cloneSliceBranch(
                incoming.root,
                mask,
                physicalTop,
                physicalBottom,
                originY,
                originY + physicalHeight
            )

            assert(
                copy,
                "Incoming view has no measurable transition bounds."
            )

            copy.AnchorPoint = Vector2.new(0, 0)
            copy.Position = UDim2.fromOffset(0, -y0)
            copy.Size = UDim2.fromOffset(T.width, T.height)
            copy.Visible = true
            copy.ZIndex = 1

            -- An opaque foundation makes overlapping slices safe.
            -- Transparency within cards and decorative surfaces
            -- remains unchanged.
            copy.BackgroundTransparency = 0
            copy.BorderSizePixel = 0

            tr.slices[#tr.slices + 1] = {
                mask = mask,
                height = maskHeight,

                -- Use the nominal band center for the diagonal,
                -- not the extended overlap region.
                middle = (y0 + y1) * 0.5,
            }
        end
    end

    -- All incoming copies now exist. Register only the animated
    -- sources that actually have copies in this transition.
    buildTransitionAnimationMirrors(tr)
    syncTransitionAnimations(tr)

    incoming.root.Visible = false

    tr.beam = make("Frame", tr.layer, {
        Name = "DiagonalBoundaryLight",
        AnchorPoint = Vector2.new(0.5, 0.5),
        Size = UDim2.fromOffset(76, T.height * 1.8 + 240),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BorderSizePixel = 0,
        Active = false,
        Interactable = false,
        ZIndex = 100,
    })

    make("UIGradient", tr.beam, {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Color3.fromRGB(130, 168, 255)),
            ColorSequenceKeypoint.new(0.40, Color3.fromRGB(216, 232, 255)),
            ColorSequenceKeypoint.new(0.50, Color3.fromRGB(255, 255, 245)),
            ColorSequenceKeypoint.new(0.60, Color3.fromRGB(255, 241, 202)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(191, 208, 246)),
        }),
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 1),
            NumberSequenceKeypoint.new(0.20, 0.97),
            NumberSequenceKeypoint.new(0.36, 0.80),
            NumberSequenceKeypoint.new(0.46, 0.32),
            NumberSequenceKeypoint.new(0.49, 0.02),
            NumberSequenceKeypoint.new(0.51, 0.02),
            NumberSequenceKeypoint.new(0.54, 0.32),
            NumberSequenceKeypoint.new(0.64, 0.80),
            NumberSequenceKeypoint.new(0.80, 0.97),
            NumberSequenceKeypoint.new(1, 1),
        }),
    })

    -- Input shield prevents interaction with visible-but-partially-clipped
    -- controls. The separate theme bar remains interactive for reversal.
    make("TextButton", tr.layer, {
        Name = "TransitionInputShield",
        Size = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1,
        Text = "",
        AutoButtonColor = false,
        Active = true,
        Selectable = false,
        ZIndex = 110,
    })

    tr.started = os.clock()
    drawTransition(tr)
    updateThemeButtons()
end

local function visualFailure(failure)
    local tr = T.transition
    if tr then T.current = tr.outgoing end
    destroyTransition()
    T.automatic = false
    updateThemeButtons()
    syncLighting()
    warn("[TDX] Theme transition stopped: " .. tostring(failure))
end

requestTheme = function(mode)
    if not S.alive or not THEME_SETTINGS.Enabled then return end
    if mode ~= "MIDNIGHT" and mode ~= "IVORY" and mode ~= "AUTO" then return end

    if mode == "AUTO" then
        T.automatic = true
        resetThemeDeadline()
        updateThemeButtons()
        return
    end

    T.automatic = false
    local tr = T.transition
    if tr then
        local goal = mode == tr.incoming and 1 or 0
        if goal ~= tr.to then
            tr.from = tr.p
            tr.to = goal
            tr.started = os.clock()
            tr.duration = math.max(0.12,
                THEME_SETTINGS.TransitionDuration * math.abs(goal - tr.p))
        end
        updateThemeButtons()
        return
    end

    resetThemeDeadline()
    local ok, failure = xpcall(function()
        beginTransition(mode)
    end, debug.traceback)
    if not ok then visualFailure(failure) end
    updateThemeButtons()
end

local function tickTheme()
    local tr = T.transition
    if tr then
        local alpha = math.clamp((os.clock() - tr.started) / tr.duration, 0, 1)
        local eased = 0.5 - math.cos(alpha * math.pi) * 0.5
        tr.p = tr.from + (tr.to - tr.from) * eased
        drawTransition(tr)
        if alpha >= 1 then settleTransition() end
    elseif THEME_SETTINGS.Enabled and T.automatic
        and not S.closing and os.clock() >= T.deadline then
        beginTransition(T.current == "MIDNIGHT" and "IVORY" or "MIDNIGHT")
    end
end

fit = function()
    if not S.alive then return end
    local size = bounds.AbsoluteSize
    if size.X <= 0 or size.Y <= 0 then return end

    local availableWidth = math.max(1, size.X - 20)
    local availableHeight = math.max(1, size.Y - 20)
    local scale = math.min(1, availableWidth / 280, availableHeight / 205)
    scale = math.max(0.01, scale)

    local width = math.min(460, availableWidth / scale)
    local totalHeight = S.collapsed and 152
        or math.min(846, availableHeight / scale)
    local height = math.max(1, totalHeight - 56)

    local dimensionsChanged = width ~= T.width or height ~= T.height or scale ~= T.scale
    if dimensionsChanged and T.transition then
        -- A viewport resize commits the currently requested destination.
        -- No stale-size masks survive into the new layout.
        settleTransition()
    end

    T.width, T.height, T.scale = width, height, scale
    shellScale.Scale = scale
    shell.Size = UDim2.fromOffset(width, totalHeight)
    shell.Position = UDim2.fromOffset(
        math.clamp(shell.Position.X.Offset, 0, math.max(0, size.X - width * scale)),
        math.clamp(shell.Position.Y.Offset, 0,
            math.max(0, size.Y - totalHeight * scale - 7))
    )

    for _, v in pairs(views) do
        v.body.Visible = not S.collapsed
        v.divider.Visible = not S.collapsed
        v.footer.Visible = not S.collapsed
        v.collapse.Text = S.collapsed and "+" or "-"
        v.reflow(width)
    end
    syncLighting()
end

cleanup = function(destroyGui)
    if not S.alive then return end
    S.alive, S.auto, S.targetAuto = false, false, false
    T.automatic = false
    dragInput = nil

    if S.outstanding then
        warn("[TDX] Removed with an unresolved request. Removal does not cancel server processing.")
    end

    for _, connection in ipairs(S.connections) do
        connection:Disconnect()
    end
    table.clear(S.connections)

    for key, tween in pairs(visualTweens) do
        visualTweens[key] = nil
        tween:Cancel()
        tween:Destroy()
    end

    local tr = T.transition
    T.transition = nil
    table.clear(T.mirrors)
    if tr and tr.layer then tr.layer:Destroy() end

    table.clear(feedbackRecords)
    table.clear(S.seen)
    table.clear(S.fresh)
    table.clear(S.targeted)
    S.job, S.uncertain = nil, nil
    local ivory = views.IVORY
    if ivory and ivory.ivory then
        table.clear(ivory.ivory.surfaces)
        ivory.ivory = nil
    end

    table.clear(views)
    table.clear(themeButtons)
    if destroyGui then gui:Destroy() end
end

requestClose = function(reason)
    if not S.alive then return end
    S.auto, S.targetAuto = false, false
    T.automatic = false
    table.clear(S.fresh)
    dragInput = nil

    outstanding()
    if S.outstanding then
        S.closing = true
        if S.job then
            S.job.cancelReason = reason or "Close requested."
        end
        message(
            S.uncertain
                and "Close queued. Outcome unknown; waiting for a level update."
                or "Close queued. Waiting for the outstanding purchase.",
            S.uncertain ~= nil
        )
        updateThemeButtons()
    else
        cleanup(true)
    end
end

dispatch = function(command, argument)
    if not S.alive then return end

    if command == "close" then
        requestClose("Close requested.")
        return
    end

    if T.transition then return end

    if command == "collapse" then
        dragInput = nil
        resetFeedback()
        S.collapsed = not S.collapsed
        fit()
        return
    end

    if command == "cancel" then
        if S.job then
            S.job.cancelReason = S.job.cancelReason or "Canceled."
            message("Stopping after any outstanding purchase settles.")
            guarded(tickJob)
        elseif S.uncertain then
            message("Outcome unknown. Cancel cannot undo a dispatched request.", true)
        elseif not S.closing then
            message("No active upgrade job.")
        end
        return
    end

    guarded(function()
        if S.closing then return end

        if command == "path" then
            if S.job or S.uncertain then return end
            if type(argument) ~= "number" or not PATHS[argument] then return end
            S.path = argument
            message(PATHS[argument].name
                .. " selected. Secondary path upgrades first.")

        elseif command == "start" then
            start(false)

        elseif command == "auto" then
            if S.uncertain then
                message("Resolve the previous outcome before enabling Auto.", true)
                return
            end

            S.auto = not S.auto
            table.clear(S.fresh)
            table.clear(S.seen)
            scanElapsed = 0

            if S.auto then
                for _, tower in pairs(API.Towers.GetTowers()) do
                    S.seen[tower] = true
                end
                local tower = current()
                if tower then S.seen[tower] = true end
            end

            message(S.auto
                and "Auto: select a newly observed owned tower within 12 seconds."
                or "Auto OFF. Existing job unchanged; Cancel stops it.")

        elseif command == "target" then
            if S.uncertain then
                message("Resolve the previous outcome before enabling targeting.", true)
                return
            end
            S.targetAuto = not S.targetAuto
            message(S.targetAuto
                and "Targeting: cycles C once per selected owned tower."
                or "Automatic targeting disabled.")
        end
    end)
end

connect(UIS.InputChanged, function(input)
    if not S.alive or not dragInput or T.transition then return end
    if input == dragInput or (
        dragInput.UserInputType == Enum.UserInputType.MouseButton1
        and input.UserInputType == Enum.UserInputType.MouseMovement
    ) then
        local delta = input.Position - dragStart
        local size = bounds.AbsoluteSize
        shell.Position = UDim2.fromOffset(
            math.clamp(
                panelStart.X.Offset + delta.X,
                0,
                math.max(0, size.X - T.width * T.scale)
            ),
            math.clamp(
                panelStart.Y.Offset + delta.Y,
                0,
                math.max(0, size.Y - shell.Size.Y.Offset * T.scale - 7)
            )
        )
    end
end)

connect(UIS.InputEnded, function(input)
    if not S.alive then return end
    if input == dragInput or (
        dragInput
        and dragInput.UserInputType == Enum.UserInputType.MouseButton1
        and input.UserInputType == Enum.UserInputType.MouseButton1

    end
end)

local uiElapsed = 0.15
local uncertainReadFailed = false
local presentationFailed = false

local function renderSafely()
    if not S.alive or presentationFailed then return end
    local ok, failure = xpcall(render, debug.traceback)
    if ok then return end

    -- Discard any broken mirrors before trying the persistent views.
    visualFailure(failure)
    local recovered, renderError = xpcall(render, debug.traceback)
    if not recovered then
        presentationFailed = true
        S.auto, S.targetAuto = false, false
        table.clear(S.fresh)
        if S.job then
            S.job.cancelReason = "Presentation stopped."
        end
        warn("[TDX] Presentation stopped: " .. tostring(renderError))
        requestClose("Presentation stopped.")
    end
end

-- One heartbeat owns gameplay progression, display sampling, and the reveal.
-- Theme animation never starts another gameplay worker.
connect(RunService.Heartbeat, function(dt)
    if not S.alive then return end

    if not S.faulted then
        guarded(function()
            if S.uncertain then
                tickUncertain()
            elseif S.job then
                tickJob()
            end

            if S.closing then return end

            tickAuto(dt)

            local ok, failure = pcall(tickTargeting)
            if not ok then
                S.targetAuto = false
                message("Automatic targeting stopped; see console.", true)
                warn("[TDX] " .. tostring(failure))
            end
        end)
    elseif S.uncertain and not uncertainReadFailed and not S.busy then
        -- A fault never authorizes another purchase. Only reconcile the
        -- request already dispatched, using the existing read-only check.
        S.busy = true
        local ok, failure = pcall(tickUncertain)
        S.busy = false
        if not ok then
            uncertainReadFailed = true
            message("Client error; outcome unknown. No retry issued.", true)
            warn("[TDX] " .. tostring(failure))
        end
    end

    if not S.alive then return end
    if S.closing and not S.outstanding and not S.busy then
        cleanup(true)
        return
    end

    uiElapsed = uiElapsed + dt
    if uiElapsed >= 0.15 then
        uiElapsed = 0
        if not S.faulted then
            guarded(sampleDisplay)
        end
        renderSafely()
    end

    if not S.alive then return end

    local ok, failure = xpcall(function()
        tickTheme()
        tickIvoryPresentation(dt)

        local tr = T.transition
        if tr then
            -- Run after Ivory has applied its material opacity values.
            -- No extra worker, tween, or connection is created.
            syncTransitionAnimations(tr)
        end

        if tr and tr.beam then
            -- Feather the light at either endpoint, including reversals.
            -- The slice masks remain opaque: this fades only the beam,
            -- never the outgoing or incoming presentation.
            local edge = math.min(tr.p, 1 - tr.p)
            tr.beam.BackgroundTransparency =
                1 - math.clamp(edge / 0.065, 0, 1)
        end
    end, debug.traceback)

    if not ok then
        visualFailure(failure)
    end
end)

-- Initialize both existing presentations before permitting automatic changes.
-- Their controls already share dispatch; no additional control connections
-- or view construction is needed here.
destroyTransition()
fit()
resetThemeDeadline()
updateThemeButtons()
guarded(sampleDisplay)
renderSafely()

if S.alive then
    syncLighting()
    if not S.faulted then
        print("[TDX] Midnight + Ivory loaded. One shared gameplay worker; "
            .. "server/client purchase limits still apply.")
    end
end
